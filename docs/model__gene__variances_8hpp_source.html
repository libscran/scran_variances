317template<typename Value_, typename Index_, typename Stat_>
318void model_gene_variances(
319 const tatami::Matrix<Value_, Index_>& mat,
320 const ModelGeneVariancesBuffers<Stat_> buffers,
321 const ModelGeneVariancesOptions& options
322) {
323 tatami_stats::VarianceBuffers<Stat_> vbuffers;
324 vbuffers.mean = buffers.means;
325 vbuffers.variance = buffers.variances;
326
327 tatami_stats::VarianceOptions vopt;
328 vopt.num_threads = options.num_threads;
329 tatami_stats::variance(true, mat, vbuffers, vopt);
330
331 FitVarianceTrendWorkspace<Stat_> work;
332 auto fopt = options.fit_variance_trend_options;
333 fopt.num_threads = options.num_threads;
334
335 if (buffers.fitted != NULL && buffers.residuals != NULL) {
336 const auto NR = mat.nrow();
337 if (mat.ncol() >= 2) {
338 fit_variance_trend(NR, buffers.means, buffers.variances, buffers.fitted, buffers.residuals, work, fopt);
339 } else {
340 std::fill_n(buffers.fitted, NR, std::numeric_limits<double>::quiet_NaN());
341 std::fill_n(buffers.residuals, NR, std::numeric_limits<double>::quiet_NaN());
342 }
343 }

371template<typename Value_, typename Index_, typename Block_, typename Stat_>
372void model_gene_variances_blocked(
373 const tatami::Matrix<Value_, Index_>& mat,
374 const Block_* const block,
375 const std::size_t num_blocks,
376 const ModelGeneVariancesBlockedBuffers<Stat_>& buffers,
377 const ModelGeneVariancesOptions& options
378) {
379 tatami_stats::GroupVarianceBuffers<Stat_> vbuffers;
380 sanisizer::resize(vbuffers.mean, num_blocks);
381 sanisizer::resize(vbuffers.variance, num_blocks);
382 for (std::size_t b = 0; b < num_blocks; ++b) {
383 vbuffers.mean[b] = buffers.per_block[b].means;
384 vbuffers.variance[b] = buffers.per_block[b].variances;
385 }
386
387 tatami_stats::GroupVarianceOptions vopt;
388 vopt.num_threads = options.num_threads;
389 tatami_stats::group_variance(true, mat, block, num_blocks, vbuffers, vopt);
390
391 FitVarianceTrendWorkspace<Stat_> work;
392 auto fopt = options.fit_variance_trend_options;
393 fopt.num_threads = options.num_threads;
394 bool all_trends_fitted = true;
395
396 const Index_ NR = mat.nrow(), NC = mat.ncol();
397 auto block_size = sanisizer::create<std::vector<Index_> >(num_blocks);
398 for (Index_ c = 0; c < NC; ++c) {
399 block_size[block[c]] += 1;
400 }
401
402 for (std::size_t b = 0; b < num_blocks; ++b) {
403 const auto& current = buffers.per_block[b];
404 if (current.fitted == NULL || current.residuals == NULL) {
405 all_trends_fitted = false;
406 continue;
407 }
408 if (block_size[b] >= 2) {
409 fit_variance_trend(NR, current.means, current.variances, current.fitted, current.residuals, work, fopt);
410 } else {
411 std::fill_n(current.fitted, NR, std::numeric_limits<double>::quiet_NaN());
412 std::fill_n(current.residuals, NR, std::numeric_limits<double>::quiet_NaN());
413 }
414 }
415
416 const auto ave_means = buffers.average.means;
417 const auto ave_variances = buffers.average.variances;
418 const auto ave_fitted = buffers.average.fitted;
419 const auto ave_residuals = buffers.average.residuals;
420
421 if ((ave_fitted || ave_residuals) && !all_trends_fitted) {
422 throw std::runtime_error("cannot compute average fitted values/residuals without per-block trend fits");
423 }
424
425 std::vector<Stat_*> tmp_pointers;
426 tmp_pointers.reserve(num_blocks);
427
428 if (options.block_average_policy == BlockAveragePolicy::MEAN) {
429 const auto block_weight = scran_blocks::compute_weights<Stat_>(block_size, options.block_weight_policy, options.variable_block_weight_parameters);
430 std::vector<Stat_> tmp_weights;
431 tmp_weights.reserve(num_blocks);
432
433 if (ave_means) {
434 internal::extract_weights(block_weight, block_size, static_cast<Index_>(1), tmp_weights);
435 internal::extract_pointers(buffers.per_block, block_size, static_cast<Index_>(1), [](const auto& x) -> Stat_* { return x.means; }, tmp_pointers);
436 scran_blocks::parallel_weighted_means(NR, tmp_pointers, tmp_weights.data(), ave_means, /* skip_nan = */ false);
437 }
438
439 // Skip blocks without enough cells to compute the variance.
440 internal::extract_weights(block_weight, block_size, static_cast<Index_>(2), tmp_weights);
441
442 if (ave_variances) {
443 internal::extract_pointers(buffers.per_block, block_size, static_cast<Index_>(2), [](const auto& x) -> Stat_* { return x.variances; }, tmp_pointers);
444 scran_blocks::parallel_weighted_means(NR, tmp_pointers, tmp_weights.data(), ave_variances, /* skip_nan = */ false);
445 }
446
447 if (ave_fitted) {
448 internal::extract_pointers(buffers.per_block, block_size, static_cast<Index_>(2), [](const auto& x) -> Stat_* { return x.fitted; }, tmp_pointers);
449 scran_blocks::parallel_weighted_means(NR, tmp_pointers, tmp_weights.data(), ave_fitted, /* skip_nan = */ false);
450 }
451
452 if (ave_residuals) {
453 internal::extract_pointers(buffers.per_block, block_size, static_cast<Index_>(2), [](const auto& x) -> Stat_* { return x.residuals; }, tmp_pointers);
454 scran_blocks::parallel_weighted_means(NR, tmp_pointers, tmp_weights.data(), ave_residuals, /* skip_nan = */ false);
455 }
456
457 } else if (options.block_average_policy == BlockAveragePolicy::QUANTILE) {
458 if (ave_means) {
459 internal::extract_pointers(buffers.per_block, block_size, static_cast<Index_>(1), [](const auto& x) -> Stat_* { return x.means; }, tmp_pointers);
460 scran_blocks::parallel_quantiles(NR, tmp_pointers, options.block_quantile, ave_means, /* skip_nan = */ false);
461 }
462
463 // Skip blocks without enough cells to compute the variance.
464
465 if (ave_variances) {
466 internal::extract_pointers(buffers.per_block, block_size, static_cast<Index_>(2), [](const auto& x) -> Stat_* { return x.variances; }, tmp_pointers);
467 scran_blocks::parallel_quantiles(NR, tmp_pointers, options.block_quantile, ave_variances, /* skip_nan = */ false);
468 }
469
470 if (ave_fitted) {
471 internal::extract_pointers(buffers.per_block, block_size, static_cast<Index_>(2), [](const auto& x) -> Stat_* { return x.fitted; }, tmp_pointers);
472 scran_blocks::parallel_quantiles(NR, tmp_pointers, options.block_quantile, ave_fitted, /* skip_nan = */ false);
473 }
474
475 if (ave_residuals) {
476 internal::extract_pointers(buffers.per_block, block_size, static_cast<Index_>(2), [](const auto& x) -> Stat_* { return x.residuals; }, tmp_pointers);
477 scran_blocks::parallel_quantiles(NR, tmp_pointers, options.block_quantile, ave_residuals, /* skip_nan = */ false);
478 }
479 }
