139template<typename Value_, typename Index_, typename Stat_>
140void model_gene_variances_direct(
141 const tatami::Matrix<Value_, Index_>& mat,
142 Stat_* const output_mean,
143 Stat_* const output_variance,
144 const int num_threads
145) {
146 const auto NR = mat.nrow();
147 const auto NC = mat.ncol();
148
149 if (mat.sparse()) {
150 tatami::Options topt;
151 topt.sparse_extract_index = false;
152
153 tatami::parallelize([&](int, Index_ s, Index_ l) -> void {
154 auto ext = tatami::consecutive_extractor<true>(mat, true, s, l, topt);
155 auto vbuffer = tatami::create_container_of_Index_size<std::vector<Value_> >(NC);
156 quickstats::RssWorkspace<Stat_> work;
157 for (Index_ x = 0; x < l; ++x) {
158 auto out = ext->fetch(vbuffer.data(), NULL);
159 const auto res = quickstats::rss(NC, out.number, out.value, work);
160 output_mean[x + s] = res.mean;
161 output_variance[x + s] = quickstats::rss_to_variance(NC, res.rss);
162 }
163 }, NR, num_threads);
164
165 } else {
166 tatami::parallelize([&](int, Index_ s, Index_ l) -> void {
167 auto ext = tatami::consecutive_extractor<false>(mat, true, s, l);
168 auto buffer = tatami::create_container_of_Index_size<std::vector<Value_> >(NC);
169 quickstats::RssWorkspace<Stat_> work;
170 for (Index_ x = 0; x < l; ++x) {
171 auto out = ext->fetch(buffer.data());
172 const auto res = quickstats::rss(NC, out, work);
173 output_mean[x + s] = res.mean;
174 output_variance[x + s] = quickstats::rss_to_variance(NC, res.rss);
175 }
176 }, NR, num_threads);
177 }

180template<typename Value_, typename Index_, typename Stat_>
181void model_gene_variances_running(
182 const tatami::Matrix<Value_, Index_>& mat,
183 Stat_* const output_mean,
184 Stat_* const output_variance,
185 const int num_threads
186) {
187 const auto NR = mat.nrow();
188 const auto NC = mat.ncol();
189 const bool is_sparse = mat.is_sparse();
190
191 const bool do_parallel = num_threads > 1;
192 std::optional<std::vector<std::optional<std::vector<Stat_> > > > all_partial_mean, all_partial_rss;
193 std::optional<std::vector<Index_> > all_partial_count;
194 if (do_parallel) {
195 // -1, as we'll repurpose the RSS output buffer to store the partial RSS of the first thread.
196 all_partial_rss.emplace(sanisizer::cast<I<decltype(all_partial_rss->size())> >(num_threads - 1));
197 all_partial_mean.emplace(sanisizer::cast<I<decltype(all_partial_mean->size())> >(num_threads));
198 all_partial_count.emplace(sanisizer::cast<I<decltype(all_partial_count->size())> >(num_threads));
199 }
200
201 std::fill_n(output_variance, NR, 0);
202 std::fill_n(output_mean, NR, 0);
203
204 const int nused = tatami::parallelize([&](int thread, Index_ s, Index_ l) -> void {
205 Stat_* rss_ptr;
206 Stat_* mean_ptr;
207 std::optional<std::vector<Stat_> > cur_rss, cur_mean;
208
209 if (!do_parallel) {
210 // Storing mean and RSS directly in the output vector to cut down two allocations if we're not working in parallel.
211 rss_ptr = output_variance;
212 mean_ptr = output_mean;
213 } else {
214 // Storing the partial RSS directly in the output vector to save ourselves an allocation if we're in the first thread.
215 // We can't do the same for the mean, though, as we need to keep the partial mean and the global mean separate for the reduction.
216 cur_mean.emplace(tatami::cast_Index_to_container_size<std::vector<Stat_> >(NR));
217 mean_ptr = cur_mean->data();
218 if (thread == 0) {
219 rss_ptr = output_variance;
220 } else {
221 cur_rss.emplace(tatami::cast_Index_to_container_size<std::vector<Stat_> >(NR));
222 rss_ptr = cur_rss->data();
223 }
224 }
225
226 if (is_sparse) {
227 tatami::Options topt;
228 topt.sparse_ordered_index = false; // we don't care about the ordering of the non-zero elements.
229 auto ext = tatami::consecutive_extractor<true>(mat, false, s, l, topt);
230 auto vbuffer = tatami::create_container_of_Index_size<std::vector<Value_> >(NR);
231 auto ibuffer = tatami::create_container_of_Index_size<std::vector<Index_> >(NR);
232 auto nonzeros = tatami::create_container_of_Index_size<std::vector<Index_> >(NR);
233
234 quickstats::RssRunningSparse<Index_, Value_, Stat_> runner(NR, mean_ptr, rss_ptr, nonzeros.data());
235 for (Index_ x = 0; x < l; ++x) {
236 auto out = ext->fetch(vbuffer.data(), ibuffer.data());
237 runner.add(out.number, out.value, out.index);
238 }
239 runner.finish();
240
241 } else {
242 auto ext = tatami::consecutive_extractor<false>(mat, false, s, l);
243 auto buffer = tatami::create_container_of_Index_size<std::vector<Value_> >(NR);
244
245 quickstats::RssRunningDense<Value_, Stat_> runner(NR, mean_ptr, rss_ptr);
246 for (Index_ x = 0; x < l; ++x) {
247 auto out = ext->fetch(buffer.data());
248 runner.add(out);
249 }
250 runner.finish();
251 }
252
253 if (do_parallel) {
254 (*all_partial_count)[thread] = l;
255 (*all_partial_mean)[thread] = std::move(cur_mean);
256 if (thread > 0) {
257 (*all_partial_rss)[thread - 1] = std::move(cur_rss);
258 }
259 }
260 }, NC, num_threads);
261
262 // Don't check nused > 1, as it's possible for do_parallel = true with nused = 1 if not all threads are used.
263 // This would cause us to leave output_mean and output_variance empty.
264 if (do_parallel) {
265 const auto& ap_count = *all_partial_count;
266 const auto& ap_mean = *all_partial_mean;
267 const auto& ap_rss = *all_partial_rss;
268
269 // Computing the global mean.
270 for (int u = 0; u < nused; ++u) {
271 const Stat_ mult = static_cast<Stat_>(ap_count[u]) / static_cast<Stat_>(NC);
272 const auto& cur_mean = *(ap_mean[u]);
273 for (Index_ d = 0; d < NR; ++d) {
274 output_mean[d] += cur_mean[d] * mult;
275 }
276 }
277
278 // Combining the RSS. We can use recenter_rss_unsafe() as we are guaranteed that cur_count > 0,
279 // as parallelize() will only ever split into non-empty ranges if those ranges are used.
280 for (int u = 0; u < nused; ++u) {
281 const auto cur_count = ap_count[u];
282 const auto& cur_mean = *(ap_mean[u]);
283 if (u == 0) {
284 for (Index_ d = 0; d < NR; ++d) {
285 output_variance[d] = quickstats::recenter_rss_unsafe(cur_count, output_variance[d], cur_mean[d], output_mean[d]);
286 }
287 } else {
288 const auto& cur_rss = *(ap_rss[u - 1]);
289 for (Index_ d = 0; d < NR; ++d) {
290 output_variance[d] += quickstats::recenter_rss_unsafe(cur_count, cur_rss[d], cur_mean[d], output_mean[d]);
291 }
292 }
293 }
294 }
295
296 if (NC == 0) {
297 // If NC == 0, nused = 0 and RssRunning::finish() doesn't even get called.
298 // So, we have to fill the output means with NaNs manually.
299 std::fill_n(output_mean, NR, std::numeric_limits<Stat_>::quiet_NaN());
300 }
301 quickstats::rss_to_variance(NR, NC, output_variance);
302}
324template<typename Value_, typename Index_, typename Stat_>
325void model_gene_variances(
326 const tatami::Matrix<Value_, Index_>& mat,
327 const ModelGeneVariancesBuffers<Stat_> buffers,
328 const ModelGeneVariancesOptions& options
329) {
330 if (mat.prefer_rows()) {
331 model_gene_variances_direct(mat, buffers.mean, buffers.variance, options.num_threads);
332 } else {

341 const auto NR = mat.nrow();
342 if (mat.ncol() >= 2) {
343 fit_variance_trend(NR, buffers.mean, buffers.variance, buffers.fitted, buffers.residual, work, fopt);
344 } else {
345 std::fill_n(buffers.fitted, NR, std::numeric_limits<double>::quiet_NaN());
346 std::fill_n(buffers.residual, NR, std::numeric_limits<double>::quiet_NaN());
347 }

508template<typename Index_, typename Stat_>
509void finalize_direct_block_means(
510 const std::size_t num_blocks,
511 const std::vector<Index_>& block_sizes,
512 std::vector<Stat_>& cur_means,
513 const Index_ i,
514 const std::vector<ModelGeneVariancesBuffers<Stat_> >& output
515) {
516 for (std::size_t b = 0; b < num_blocks; ++b) {
517 auto& mean = cur_means[b];
518 if (block_sizes[b] == 0) {

521 mean /= block_sizes[b];

527template<typename Value_, typename Index_, typename Block_, typename Stat_>
528void model_gene_variances_blocked_direct(
529 const tatami::Matrix<Value_, Index_>& mat,
530 const Block_* const block,
531 const std::size_t num_blocks,
532 const std::vector<Index_>& block_sizes,
533 const std::vector<ModelGeneVariancesBuffers<Stat_> >& output,
534 int num_threads
535) {
536 const auto NR = mat.nrow(), NC = mat.ncol();
537 assert(sanisizer::is_equal(num_blocks, output.size()));
538 assert(sanisizer::is_equal(num_blocks, block_sizes.size()));
539
540 if (mat.sparse()) {
541 tatami::parallelize([&](int, Index_ s, Index_ l) -> void {
542 auto ext = tatami::consecutive_extractor<true>(mat, true, s, l);
543 auto vbuffer = tatami::create_container_of_Index_size<std::vector<Value_> >(NC);
544 auto ibuffer = tatami::create_container_of_Index_size<std::vector<Index_> >(NC);
545 auto cur_means = sanisizer::create<std::vector<Stat_> >(num_blocks);
546 auto cur_rss = sanisizer::create<std::vector<Stat_> >(num_blocks);
547 auto cur_non_zeros = sanisizer::create<std::vector<Index_> >(num_blocks);
548
549 for (Index_ x = 0; x < l; ++x) {
550 auto range = ext->fetch(vbuffer.data(), ibuffer.data());
551
552 // Computing the mean first.
553 for (Index_ i = 0; i < range.number; ++i) {
554 const auto g = block[range.index[i]];
555 cur_means[g] += range.value[i];
556 ++cur_non_zeros[g];
557 }
558 finalize_direct_block_means(num_blocks, block_sizes, cur_means, static_cast<Index_>(s + x), output);
559
560 // Now computing the RSS.
561 for (Index_ i = 0; i < range.number; ++i) {
562 const auto g = block[range.index[i]];
563 const auto delta = range.value[i] - cur_means[g];
564 cur_rss[g] += delta * delta;
565 }
566 for (std::size_t b = 0; b < num_blocks; ++b) {
567 const Stat_ my_rss = cur_rss[b] + cur_means[b] * cur_means[b] * (block_sizes[b] - cur_non_zeros[b]);
568 output[b].variance[s + x] = quickstats::rss_to_variance(block_sizes[b], my_rss);
569 }
570
571 std::fill(cur_means.begin(), cur_means.end(), 0);
572 std::fill(cur_rss.begin(), cur_rss.end(), 0);
573 std::fill(cur_non_zeros.begin(), cur_non_zeros.end(), 0);
574 }
575 }, NR, num_threads);
576
577 } else {
578 tatami::parallelize([&](int, Index_ s, Index_ l) -> void {
579 auto ext = tatami::consecutive_extractor<false>(mat, true, s, l);
580 auto buffer = tatami::create_container_of_Index_size<std::vector<Value_> >(NC);
581 auto cur_means = sanisizer::create<std::vector<Stat_> >(num_blocks);
582 auto cur_rss = sanisizer::create<std::vector<Stat_> >(num_blocks);
583
584 for (Index_ x = 0; x < l; ++x) {
585 auto ptr = ext->fetch(buffer.data());
586
587 // Computing the mean first.
588 for (Index_ j = 0; j < NC; ++j) {
589 cur_means[block[j]] += ptr[j];
590 }
591 finalize_direct_block_means(num_blocks, block_sizes, cur_means, static_cast<Index_>(s + x), output);
592
593 // Now computing the RSS.
594 for (Index_ j = 0; j < NC; ++j) {
595 const auto g = block[j];
596 const auto delta = ptr[j] - cur_means[g];
597 cur_rss[g] += delta * delta;
598 }
599 for (std::size_t b = 0; b < num_blocks; ++b) {
600 output[b].variance[s + x] = quickstats::rss_to_variance(block_sizes[b], cur_rss[b]);
601 }
602
603 std::fill(cur_means.begin(), cur_means.end(), 0);
604 std::fill(cur_rss.begin(), cur_rss.end(), 0);
605 }
606 }, NR, num_threads);
607 }
608}

610template<typename Value_, typename Index_, typename Group_, typename Stat_>
611void model_gene_variances_blocked_running(
612 const tatami::Matrix<Value_, Index_>& mat,
613 const Group_* const block,
614 const std::size_t num_blocks,
615 const std::vector<Index_>& block_sizes,
616 const std::vector<ModelGeneVariancesBuffers<Stat_> >& output,
617 int num_threads
618) {
619 const auto NR = mat.nrow(), NC = mat.ncol();
620 const bool is_sparse = mat.is_sparse();
621
622 const bool do_parallel = num_threads > 1;
623 std::optional<std::vector<std::optional<std::vector<std::vector<Stat_> > > > > all_partial_mean, all_partial_rss;
624 std::optional<std::vector<std::optional<std::vector<Index_> > > > all_partial_count;
625 if (do_parallel) {
626 // -1, as we'll repurpose the RSS output buffer to store the partial RSS of the first thread.
627 all_partial_rss.emplace(sanisizer::cast<I<decltype(all_partial_rss->size())> >(num_threads - 1));
628 all_partial_mean.emplace(sanisizer::cast<I<decltype(all_partial_mean->size())> >(num_threads));
629 all_partial_count.emplace(sanisizer::cast<I<decltype(all_partial_count->size())> >(num_threads));
630 }
631
632 assert(sanisizer::is_equal(num_blocks, output.size()));
633 for (std::size_t b = 0; b < num_blocks; ++b) {
634 std::fill_n(output[b].mean, NR, 0);
635 std::fill_n(output[b].variance, NR, 0);
636 }
637
638 const int nused = tatami::parallelize([&](int thread, Index_ s, Index_ l) -> void {
639 auto tmp_mean_ptrs = sanisizer::create<std::vector<Stat_*> >(num_blocks);
640 auto tmp_rss_ptrs = sanisizer::create<std::vector<Stat_*> >(num_blocks);
641 std::optional<std::vector<std::vector<Stat_> > > cur_mean, cur_rss;
642
643 if (!do_parallel) {
644 // Storing mean and RSS directly in the output vector to cut down two allocations if we're not working in parallel.
645 for (std::size_t b = 0; b < num_blocks; ++b) {
646 tmp_mean_ptrs[b] = output[b].mean;
647 tmp_rss_ptrs[b] = output[b].variance;
648 }
649
650 } else {
651 // Storing the partial RSS directly in the output vectors to save ourselves an allocation if we're in the first thread.
652 // We can't do the same for the mean, though, as we need to keep the partial mean and the global mean separate for the reduction.
653 cur_mean.emplace(sanisizer::cast<I<decltype(cur_mean->size())> >(num_blocks));
654 for (std::size_t b = 0; b < num_blocks; ++b) {
655 tatami::resize_container_to_Index_size((*cur_mean)[b], NR);
656 tmp_mean_ptrs[b] = (*cur_mean)[b].data();
657 }
658
659 if (thread == 0) {
660 for (std::size_t b = 0; b < num_blocks; ++b) {
661 tmp_rss_ptrs[b] = output[b].variance;
662 }
663 } else {
664 cur_rss.emplace(sanisizer::cast<I<decltype(cur_rss->size())> >(num_blocks));
665 for (std::size_t b = 0; b < num_blocks; ++b) {
666 tatami::resize_container_to_Index_size((*cur_rss)[b], NR);
667 tmp_rss_ptrs[b] = (*cur_rss)[b].data();
668 }
669 }
670 }
671
672 if (is_sparse) {
673 auto ext = tatami::consecutive_extractor<true>(mat, false, s, l);
674 auto vbuffer = tatami::create_container_of_Index_size<std::vector<Value_> >(NR);
675 auto ibuffer = tatami::create_container_of_Index_size<std::vector<Index_> >(NR);
676
677 auto nonzeros = sanisizer::create<std::vector<std::vector<Index_> > >(num_blocks);
678 std::vector<quickstats::RssRunningSparse<Index_, Value_, Stat_> > runners;
679 sanisizer::reserve(runners, num_blocks);
680 for (std::size_t b = 0; b < num_blocks; ++b) {
681 tatami::resize_container_to_Index_size(nonzeros[b], NR);
682 runners.emplace_back(NR, tmp_mean_ptrs[b], tmp_rss_ptrs[b], nonzeros[b].data());
683 }
684
685 for (Index_ x = 0; x < l; ++x) {
686 auto out = ext->fetch(vbuffer.data(), ibuffer.data());
687 runners[block[x + s]].add(out.number, out.value, out.index);
688 }
689
690 for (std::size_t b = 0; b < num_blocks; ++b) {
691 runners[b].finish();
692 }
693
694 } else {
695 auto ext = tatami::consecutive_extractor<false>(mat, false, s, l);
696 auto buffer = tatami::create_container_of_Index_size<std::vector<Value_> >(NR);
697
698 std::vector<quickstats::RssRunningDense<Value_, Stat_> > runners;
699 sanisizer::reserve(runners, num_blocks);
700 for (std::size_t b = 0; b < num_blocks; ++b) {
701 runners.emplace_back(NR, tmp_mean_ptrs[b], tmp_rss_ptrs[b]);
702 }
703
704 for (Index_ x = 0; x < l; ++x) {
705 auto out = ext->fetch(buffer.data());
706 runners[block[x + s]].add(out);
707 }
708
709 for (std::size_t b = 0; b < num_blocks; ++b) {
710 runners[b].finish();
711 }
712 }
713
714 if (do_parallel) {
715 auto cur_count = sanisizer::create<std::vector<Index_> >(num_blocks);
716 for (Index_ x = 0; x < l; ++x) {
717 cur_count[block[x + s]] += 1;
718 }
719 (*all_partial_count)[thread] = std::move(cur_count);
720
721 (*all_partial_mean)[thread] = std::move(cur_mean);
722 if (thread > 0) {
723 (*all_partial_rss)[thread - 1] = std::move(cur_rss);
724 }
725 }
726 }, NC, num_threads);
727
728 if (!do_parallel) {
729 for (std::size_t b = 0; b < num_blocks; ++b) {
730 quickstats::rss_to_variance(NR, block_sizes[b], output[b].variance);
731 }
732 if (NC == 0) {
733 // If NC == 0, nused = 0 and RssRunning::finish() doesn't even get called.
734 // So, we have to fill the output means with NaNs manually.
735 for (std::size_t b = 0; b < num_blocks; ++b) {
736 std::fill_n(output[b].mean, NR, std::numeric_limits<Stat_>::quiet_NaN());
737 }
738 }
739
740 } else {
741 const auto& ap_mean = *all_partial_mean;
742 const auto& ap_rss = *all_partial_rss;
743
744 // Computing the global mean.
745 assert(sanisizer::is_equal(num_blocks, block_sizes.size()));
746 for (std::size_t b = 0; b < num_blocks; ++b) {
747 const auto cur_output = output[b].mean;
748 if (block_sizes[b] == 0) {
749 std::fill_n(cur_output, NR, std::numeric_limits<Stat_>::quiet_NaN());
750 continue;
751 }
752 for (int u = 0; u < nused; ++u) {
753 const auto cur_count = (*((*all_partial_count)[u]))[b];
754 if (cur_count == 0) {
755 continue;
756 }
757 const auto& cur_mean = (*(ap_mean[u]))[b];
758 const Stat_ mult = static_cast<Stat_>(cur_count) / static_cast<Stat_>(block_sizes[b]);
759 for (Index_ d = 0; d < NR; ++d) {
760 cur_output[d] += cur_mean[d] * mult;
761 }
762 }
763 }
764
765 // Combining the RSS.
766 for (std::size_t b = 0; b < num_blocks; ++b) {
767 const auto& cur_global = output[b].mean;
768 const auto cur_output = output[b].variance;
769 for (int u = 0; u < nused; ++u) {
770 const auto cur_count = (*((*all_partial_count)[u]))[b];
771 if (cur_count == 0) {
772 continue;
773 }
774 const auto& cur_mean = (*(ap_mean[u]))[b];
775 if (u == 0) {
776 for (Index_ d = 0; d < NR; ++d) {
777 // At this point, we know cur_count > 0, so we can use the unsafe function.
778 cur_output[d] = quickstats::recenter_rss_unsafe(cur_count, cur_output[d], cur_mean[d], cur_global[d]);
779 }
780 } else {
781 const auto& cur_rss = (*(ap_rss[u - 1]))[b];
782 for (Index_ d = 0; d < NR; ++d) {
783 cur_output[d] += quickstats::recenter_rss_unsafe(cur_count, cur_rss[d], cur_mean[d], cur_global[d]);
784 }
785 }
786 }
787 quickstats::rss_to_variance(NR, block_sizes[b], cur_output);
788 }
789 }
790}

859template<typename Value_, typename Index_, typename Block_, typename Stat_>
860void model_gene_variances_blocked(
861 const tatami::Matrix<Value_, Index_>& mat,
862 const Block_* const block,
863 const std::size_t num_blocks,
864 const ModelGeneVariancesBlockedBuffers<Stat_>& buffers,
865 const ModelGeneVariancesOptions& options
866) {
867 if (!sanisizer::is_equal(num_blocks, buffers.per_block.size())) {
868 throw std::runtime_error("length of 'buffers.per_block' is not equal to 'num_blocks'");
869 }
870 assert(mat.ncol() == 0 || sanisizer::is_less_than(*std::max_element(block, block + mat.ncol()), num_blocks));
871
872 // Just compute the block sizes here for simplicity.
873 auto block_sizes = sanisizer::create<std::vector<Index_> >(num_blocks);
874 const auto NC = mat.ncol();
875 for (Index_ c = 0; c < NC; ++c) {
876 block_sizes[block[c]] += 1;
877 }
878
879 if (mat.prefer_rows()) {
880 model_gene_variances_blocked_direct(mat, block, num_blocks, block_sizes, buffers.per_block, options.num_threads);
881 } else {
882 model_gene_variances_blocked_running(mat, block, num_blocks, block_sizes, buffers.per_block, options.num_threads);
883 }
884
885 FitVarianceTrendWorkspace<Stat_> work;
886 auto fopt = options.fit_variance_trend_options;
887 fopt.num_threads = options.num_threads;
888 bool all_trends_fitted = true;
889
890 const auto NR = mat.nrow();
891 for (std::size_t b = 0; b < num_blocks; ++b) {
892 const auto& current = buffers.per_block[b];
893 if (current.fitted == NULL || current.residual == NULL) {
894 all_trends_fitted = false;
895 continue;
896 }
897 if (block_sizes[b] >= 2) {
898 fit_variance_trend(NR, current.mean, current.variance, current.fitted, current.residual, work, fopt);
899 } else {
900 std::fill_n(current.fitted, NR, std::numeric_limits<double>::quiet_NaN());
901 std::fill_n(current.residual, NR, std::numeric_limits<double>::quiet_NaN());
902 }
903 }
904
905 const auto ave_means = buffers.average.mean;
906 const auto ave_variances = buffers.average.variance;
907 const auto ave_fitted = buffers.average.fitted;
908 const auto ave_residuals = buffers.average.residual;
909
910 if ((ave_fitted || ave_residuals) && !all_trends_fitted) {
911 throw std::runtime_error("cannot compute average fitted values/residuals without per-block trend fits");
912 }
913
914 std::vector<Stat_*> tmp_pointers;
915 tmp_pointers.reserve(num_blocks);
916
917 if (options.block_average_policy == BlockAveragePolicy::MEAN) {
918 const auto block_weight = scran_blocks::compute_weights<Stat_>(block_sizes, options.block_weight_policy, options.variable_block_weight_parameters);
919 std::vector<Stat_> tmp_weights;
920 tmp_weights.reserve(num_blocks);
921
922 if (ave_means) {
923 extract_blocked_weights(num_blocks, block_weight, block_sizes, static_cast<Index_>(1), tmp_weights);
924 extract_blocked_pointers(num_blocks, buffers.per_block, block_sizes, static_cast<Index_>(1), [](const auto& x) -> Stat_* { return x.mean; }, tmp_pointers);
925 scran_blocks::parallel_weighted_means(NR, tmp_pointers, tmp_weights.data(), ave_means, /* skip_nan = */ false);
926 }
927
928 // Skip blocks without enough cells to compute the variance.
929 extract_blocked_weights(num_blocks, block_weight, block_sizes, static_cast<Index_>(2), tmp_weights);
930
931 if (ave_variances) {
932 extract_blocked_pointers(num_blocks, buffers.per_block, block_sizes, static_cast<Index_>(2), [](const auto& x) -> Stat_* { return x.variance; }, tmp_pointers);
933 scran_blocks::parallel_weighted_means(NR, tmp_pointers, tmp_weights.data(), ave_variances, /* skip_nan = */ false);
934 }
935
936 if (ave_fitted) {
937 extract_blocked_pointers(num_blocks, buffers.per_block, block_sizes, static_cast<Index_>(2), [](const auto& x) -> Stat_* { return x.fitted; }, tmp_pointers);
938 scran_blocks::parallel_weighted_means(NR, tmp_pointers, tmp_weights.data(), ave_fitted, /* skip_nan = */ false);
939 }
940
941 if (ave_residuals) {
942 extract_blocked_pointers(num_blocks, buffers.per_block, block_sizes, static_cast<Index_>(2), [](const auto& x) -> Stat_* { return x.residual; }, tmp_pointers);
943 scran_blocks::parallel_weighted_means(NR, tmp_pointers, tmp_weights.data(), ave_residuals, /* skip_nan = */ false);
944 }
945
946 } else if (options.block_average_policy == BlockAveragePolicy::QUANTILE) {
947 if (ave_means) {
948 extract_blocked_pointers(num_blocks, buffers.per_block, block_sizes, static_cast<Index_>(1), [](const auto& x) -> Stat_* { return x.mean; }, tmp_pointers);
949 scran_blocks::parallel_quantiles(NR, tmp_pointers, options.block_quantile, ave_means, /* skip_nan = */ false);
950 }
951
952 // Again, skip blocks without enough cells to compute the variance.
953
954 if (ave_variances) {
955 extract_blocked_pointers(num_blocks, buffers.per_block, block_sizes, static_cast<Index_>(2), [](const auto& x) -> Stat_* { return x.variance; }, tmp_pointers);
956 scran_blocks::parallel_quantiles(NR, tmp_pointers, options.block_quantile, ave_variances, /* skip_nan = */ false);
957 }
958
959 if (ave_fitted) {
960 extract_blocked_pointers(num_blocks, buffers.per_block, block_sizes, static_cast<Index_>(2), [](const auto& x) -> Stat_* { return x.fitted; }, tmp_pointers);
961 scran_blocks::parallel_quantiles(NR, tmp_pointers, options.block_quantile, ave_fitted, /* skip_nan = */ false);
962 }
963
964 if (ave_residuals) {
965 extract_blocked_pointers(num_blocks, buffers.per_block, block_sizes, static_cast<Index_>(2), [](const auto& x) -> Stat_* { return x.residual; }, tmp_pointers);
966 scran_blocks::parallel_quantiles(NR, tmp_pointers, options.block_quantile, ave_residuals, /* skip_nan = */ false);
967 }
968 }
969}
