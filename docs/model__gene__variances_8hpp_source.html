138template<typename Value_, typename Index_, typename Stat_>
139void model_gene_variances_direct(
140 const tatami::Matrix<Value_, Index_>& mat,
141 Stat_* const output_mean,
142 Stat_* const output_variance,
143 const int num_threads
144) {
145 const auto NR = mat.nrow();
146 const auto NC = mat.ncol();
147
148 if (mat.sparse()) {
149 tatami::Options topt;
150 topt.sparse_extract_index = false;
151
152 tatami::parallelize([&](int, Index_ s, Index_ l) -> void {
153 auto ext = tatami::consecutive_extractor<true>(mat, true, s, l, topt);
154 auto vbuffer = tatami::create_container_of_Index_size<std::vector<Value_> >(NC);
155 quickstats::RssWorkspace<Stat_> work;
156 for (Index_ x = 0; x < l; ++x) {
157 auto out = ext->fetch(vbuffer.data(), NULL);
158 const auto res = quickstats::rss(NC, out.number, out.value, work);
159 output_mean[x + s] = res.mean;
160 output_variance[x + s] = quickstats::rss_to_variance(NC, res.rss);
161 }
162 }, NR, num_threads);
163
164 } else {
165 tatami::parallelize([&](int, Index_ s, Index_ l) -> void {
166 auto ext = tatami::consecutive_extractor<false>(mat, true, s, l);
167 auto buffer = tatami::create_container_of_Index_size<std::vector<Value_> >(NC);
168 quickstats::RssWorkspace<Stat_> work;
169 for (Index_ x = 0; x < l; ++x) {
170 auto out = ext->fetch(buffer.data());
171 const auto res = quickstats::rss(NC, out, work);
172 output_mean[x + s] = res.mean;
173 output_variance[x + s] = quickstats::rss_to_variance(NC, res.rss);
174 }
175 }, NR, num_threads);
176 }
177}

179template<typename Value_, typename Index_, typename Stat_>
180void model_gene_variances_running(
181 const tatami::Matrix<Value_, Index_>& mat,
182 Stat_* const output_mean,
183 Stat_* const output_variance,
184 const int num_threads
185) {
186 const auto NR = mat.nrow();
187 const auto NC = mat.ncol();
188 const bool is_sparse = mat.is_sparse();
189
190 const bool do_parallel = num_threads > 1;
191 std::optional<std::vector<std::optional<std::vector<Stat_> > > > all_partial_mean, all_partial_rss;
192 std::optional<std::vector<Index_> > all_partial_count;
193 if (do_parallel) {
194 // -1, as we'll repurpose the RSS output buffer to store the partial RSS of the first thread.
195 all_partial_rss.emplace(sanisizer::cast<I<decltype(all_partial_rss->size())> >(num_threads - 1));
196 all_partial_mean.emplace(sanisizer::cast<I<decltype(all_partial_mean->size())> >(num_threads));
197 all_partial_count.emplace(sanisizer::cast<I<decltype(all_partial_count->size())> >(num_threads));
198 }
199
200 std::fill_n(output_variance, NR, 0);
201 std::fill_n(output_mean, NR, 0);
202
203 const int nused = tatami::parallelize([&](int thread, Index_ s, Index_ l) -> void {
204 Stat_* rss_ptr;
205 Stat_* mean_ptr;
206 std::optional<std::vector<Stat_> > cur_rss, cur_mean;
207
208 if (!do_parallel) {
209 // Storing mean and RSS directly in the output vector to cut down two allocations if we're not working in parallel.
210 rss_ptr = output_variance;
211 mean_ptr = output_mean;
212 } else {
213 // Storing the partial RSS directly in the output vector to save ourselves an allocation if we're in the first thread.
214 // We can't do the same for the mean, though, as we need to keep the partial mean and the global mean separate for the reduction.
215 cur_mean.emplace(tatami::cast_Index_to_container_size<std::vector<Stat_> >(NR));
216 mean_ptr = cur_mean->data();
217 if (thread == 0) {
218 rss_ptr = output_variance;
219 } else {
220 cur_rss.emplace(tatami::cast_Index_to_container_size<std::vector<Stat_> >(NR));
221 rss_ptr = cur_rss->data();
222 }
223 }
224
225 if (is_sparse) {
226 tatami::Options topt;
227 topt.sparse_ordered_index = false; // we don't care about the ordering of the non-zero elements.
228 auto ext = tatami::consecutive_extractor<true>(mat, false, s, l, topt);
229 auto vbuffer = tatami::create_container_of_Index_size<std::vector<Value_> >(NR);
230 auto ibuffer = tatami::create_container_of_Index_size<std::vector<Index_> >(NR);
231 auto nonzeros = tatami::create_container_of_Index_size<std::vector<Index_> >(NR);
232
233 quickstats::RssRunningSparse<Index_, Value_, Stat_> runner(NR, mean_ptr, rss_ptr, nonzeros.data());
234 for (Index_ x = 0; x < l; ++x) {
235 auto out = ext->fetch(vbuffer.data(), ibuffer.data());
236 runner.add(out.number, out.value, out.index);
237 }
238 runner.finish();
239
240 } else {
241 auto ext = tatami::consecutive_extractor<false>(mat, false, s, l);
242 auto buffer = tatami::create_container_of_Index_size<std::vector<Value_> >(NR);
243
244 quickstats::RssRunningDense<Value_, Stat_> runner(NR, mean_ptr, rss_ptr);
245 for (Index_ x = 0; x < l; ++x) {
246 auto out = ext->fetch(buffer.data());
247 runner.add(out);
248 }
249 runner.finish();
250 }
251
252 if (do_parallel) {
253 (*all_partial_count)[thread] = l;
254 (*all_partial_mean)[thread] = std::move(cur_mean);
255 if (thread > 0) {
256 (*all_partial_rss)[thread - 1] = std::move(cur_rss);
257 }
258 }
259 }, NC, num_threads);
260
261 // Don't check nused > 1, as it's possible for do_parallel = true with nused = 1 if not all threads are used.
262 // This would cause us to leave output_mean and output_variance empty.
263 if (do_parallel) {
264 const auto& ap_count = *all_partial_count;
265 const auto& ap_mean = *all_partial_mean;
266 const auto& ap_rss = *all_partial_rss;
267
268 // Computing the global mean.
269 for (int u = 0; u < nused; ++u) {
270 const Stat_ mult = static_cast<Stat_>(ap_count[u]) / static_cast<Stat_>(NC);
271 const auto& cur_mean = *(ap_mean[u]);
272 for (Index_ d = 0; d < NR; ++d) {
273 output_mean[d] += cur_mean[d] * mult;
274 }
275 }
276
277 // Combining the RSS. We can use recenter_rss_unsafe() as we are guaranteed that cur_count > 0,
278 // as parallelize() will only ever split into non-empty ranges if those ranges are used.
279 for (int u = 0; u < nused; ++u) {
280 const auto cur_count = ap_count[u];
281 const auto& cur_mean = *(ap_mean[u]);
282 if (u == 0) {
283 for (Index_ d = 0; d < NR; ++d) {
284 output_variance[d] = quickstats::recenter_rss_unsafe(cur_count, output_variance[d], cur_mean[d], output_mean[d]);
285 }
286 } else {
287 const auto& cur_rss = *(ap_rss[u - 1]);
288 for (Index_ d = 0; d < NR; ++d) {
289 output_variance[d] += quickstats::recenter_rss_unsafe(cur_count, cur_rss[d], cur_mean[d], output_mean[d]);
290 }
291 }
292 }
293 }
294
295 quickstats::rss_to_variance(NR, NC, output_variance);
296}
318template<typename Value_, typename Index_, typename Stat_>
319void model_gene_variances(
320 const tatami::Matrix<Value_, Index_>& mat,
321 const ModelGeneVariancesBuffers<Stat_> buffers,
322 const ModelGeneVariancesOptions& options
323) {
324 if (mat.prefer_rows()) {
325 model_gene_variances_direct(mat, buffers.mean, buffers.variance, options.num_threads);
326 } else {
327 model_gene_variances_running(mat, buffers.mean, buffers.variance, options.num_threads);
328 }
329
330 FitVarianceTrendWorkspace<Stat_> work;
331 auto fopt = options.fit_variance_trend_options;
332 fopt.num_threads = options.num_threads;
333
334 if (buffers.fitted != NULL && buffers.residual != NULL) {
335 const auto NR = mat.nrow();
336 if (mat.ncol() >= 2) {
337 fit_variance_trend(NR, buffers.mean, buffers.variance, buffers.fitted, buffers.residual, work, fopt);
338 } else {
339 std::fill_n(buffers.fitted, NR, std::numeric_limits<double>::quiet_NaN());
340 std::fill_n(buffers.residual, NR, std::numeric_limits<double>::quiet_NaN());
341 }
342 }

504 const std::size_t num_blocks,
505 const std::vector<Index_>& block_sizes,

521template<typename Value_, typename Index_, typename Block_, typename Stat_>
522void model_gene_variances_blocked_direct(
523 const tatami::Matrix<Value_, Index_>& mat,
524 const Block_* const block,
525 const std::size_t num_blocks,
526 std::vector<Index_>& block_sizes,
527 const std::vector<ModelGeneVariancesBuffers<Stat_> >& output,
528 int num_threads
529) {
530 const auto NR = mat.nrow(), NC = mat.ncol();
531 assert(sanisizer::is_equal(num_blocks, output.size()));
532 assert(sanisizer::is_equal(num_blocks, block_sizes.size()));
533 for (Index_ c = 0; c < NC; ++c) {
534 block_sizes[block[c]] += 1;
535 }
536
537 if (mat.sparse()) {
538 tatami::parallelize([&](int, Index_ s, Index_ l) -> void {
539 auto ext = tatami::consecutive_extractor<true>(mat, true, s, l);
540 auto vbuffer = tatami::create_container_of_Index_size<std::vector<Value_> >(NC);
541 auto ibuffer = tatami::create_container_of_Index_size<std::vector<Index_> >(NC);
542 auto cur_means = sanisizer::create<std::vector<Stat_> >(num_blocks);
543 auto cur_rss = sanisizer::create<std::vector<Stat_> >(num_blocks);
544 auto cur_non_zeros = sanisizer::create<std::vector<Index_> >(num_blocks);
545
546 for (Index_ x = 0; x < l; ++x) {
547 auto range = ext->fetch(vbuffer.data(), ibuffer.data());
548
549 // Computing the mean first.
550 for (Index_ i = 0; i < range.number; ++i) {
551 const auto g = block[range.index[i]];
552 cur_means[g] += range.value[i];
553 ++cur_non_zeros[g];
554 }
555 finalize_direct_block_means(num_blocks, block_sizes, cur_means, static_cast<Index_>(s + x), output);
556
557 // Now computing the RSS.
558 for (Index_ i = 0; i < range.number; ++i) {
559 const auto g = block[range.index[i]];
560 const auto delta = range.value[i] - cur_means[g];
561 cur_rss[g] += delta * delta;
562 }
563 for (std::size_t b = 0; b < num_blocks; ++b) {
564 const Stat_ my_rss = cur_rss[b] + cur_means[b] * cur_means[b] * (block_sizes[b] - cur_non_zeros[b]);
565 output[b].variance[s + x] = quickstats::rss_to_variance(block_sizes[b], my_rss);
566 }
567
568 std::fill(cur_means.begin(), cur_means.end(), 0);
569 std::fill(cur_rss.begin(), cur_rss.end(), 0);
570 std::fill(cur_non_zeros.begin(), cur_non_zeros.end(), 0);
571 }
572 }, NR, num_threads);
573
574 } else {
575 tatami::parallelize([&](int, Index_ s, Index_ l) -> void {
576 auto ext = tatami::consecutive_extractor<false>(mat, true, s, l);
577 auto buffer = tatami::create_container_of_Index_size<std::vector<Value_> >(NC);
578 auto cur_means = sanisizer::create<std::vector<Stat_> >(num_blocks);
579 auto cur_rss = sanisizer::create<std::vector<Stat_> >(num_blocks);
580
581 for (Index_ x = 0; x < l; ++x) {
582 auto ptr = ext->fetch(buffer.data());
583
584 // Computing the mean first.
585 for (Index_ j = 0; j < NC; ++j) {
586 cur_means[block[j]] += ptr[j];
587 }
588 finalize_direct_block_means(num_blocks, block_sizes, cur_means, static_cast<Index_>(s + x), output);
589
590 // Now computing the RSS.
591 for (Index_ j = 0; j < NC; ++j) {
592 const auto g = block[j];
593 const auto delta = ptr[j] - cur_means[g];
594 cur_rss[g] += delta * delta;
595 }
596 for (std::size_t b = 0; b < num_blocks; ++b) {
597 output[b].variance[s + x] = quickstats::rss_to_variance(block_sizes[b], cur_rss[b]);
598 }
599
600 std::fill(cur_means.begin(), cur_means.end(), 0);
601 std::fill(cur_rss.begin(), cur_rss.end(), 0);
602 }
603 }, NR, num_threads);
604 }
605}

607template<typename Value_, typename Index_, typename Group_, typename Stat_>
608void model_gene_variances_blocked_running(
609 const tatami::Matrix<Value_, Index_>& mat,
610 const Group_* const block,
611 const std::size_t num_blocks,
612 std::vector<Index_>& block_sizes,
613 const std::vector<ModelGeneVariancesBuffers<Stat_> >& output,
614 int num_threads
615) {
616 const auto NR = mat.nrow(), NC = mat.ncol();
617 const bool is_sparse = mat.is_sparse();
618
619 const bool do_parallel = num_threads > 1;
620 std::optional<std::vector<std::optional<std::vector<std::vector<Stat_> > > > > all_partial_mean, all_partial_rss;
621 if (do_parallel) {
622 // -1, as we'll repurpose the RSS output buffer to store the partial RSS of the first thread.
623 all_partial_rss.emplace(sanisizer::cast<I<decltype(all_partial_mean->size())> >(num_threads - 1));
624 all_partial_mean.emplace(sanisizer::cast<I<decltype(all_partial_mean->size())> >(num_threads));
625 }
626 auto all_partial_count = sanisizer::create<std::vector<std::optional<std::vector<Index_> > > >(num_threads);
627
628 assert(sanisizer::is_equal(num_blocks, output.size()));
629 for (std::size_t b = 0; b < num_blocks; ++b) {
630 std::fill_n(output[b].mean, NR, 0);
631 std::fill_n(output[b].variance, NR, 0);
632 }
633
634 const int nused = tatami::parallelize([&](int thread, Index_ s, Index_ l) -> void {
635 auto tmp_mean_ptrs = sanisizer::create<std::vector<Stat_*> >(num_blocks);
636 auto tmp_rss_ptrs = sanisizer::create<std::vector<Stat_*> >(num_blocks);
637 std::optional<std::vector<std::vector<Stat_> > > cur_mean, cur_rss;
638
639 if (!do_parallel) {
640 // Storing mean and RSS directly in the output vector to cut down two allocations if we're not working in parallel.
641 for (std::size_t b = 0; b < num_blocks; ++b) {
642 tmp_mean_ptrs[b] = output[b].mean;
643 tmp_rss_ptrs[b] = output[b].variance;
644 }
645
646 } else {
647 // Storing the partial RSS directly in the output vectors to save ourselves an allocation if we're in the first thread.
648 // We can't do the same for the mean, though, as we need to keep the partial mean and the global mean separate for the reduction.
649 cur_mean.emplace(sanisizer::cast<I<decltype(cur_mean->size())> >(num_blocks));
650 for (std::size_t b = 0; b < num_blocks; ++b) {
651 tatami::resize_container_to_Index_size((*cur_mean)[b], NR);
652 tmp_mean_ptrs[b] = (*cur_mean)[b].data();
653 }
654
655 if (thread == 0) {
656 for (std::size_t b = 0; b < num_blocks; ++b) {
657 tmp_rss_ptrs[b] = output[b].variance;
658 }
659 } else {
660 cur_rss.emplace(sanisizer::cast<I<decltype(cur_rss->size())> >(num_blocks));
661 for (std::size_t b = 0; b < num_blocks; ++b) {
662 tatami::resize_container_to_Index_size((*cur_rss)[b], NR);
663 tmp_rss_ptrs[b] = (*cur_rss)[b].data();
664 }
665 }
666 }
667
668 auto cur_count = sanisizer::create<std::vector<Index_> >(num_blocks);
669 for (Index_ x = 0; x < l; ++x) {
670 cur_count[block[x + s]] += 1;
671 }
672
673 if (is_sparse) {
674 auto ext = tatami::consecutive_extractor<true>(mat, false, s, l);
675 auto vbuffer = tatami::create_container_of_Index_size<std::vector<Value_> >(NR);
676 auto ibuffer = tatami::create_container_of_Index_size<std::vector<Index_> >(NR);
677
678 auto nonzeros = sanisizer::create<std::vector<std::vector<Index_> > >(num_blocks);
679 std::vector<quickstats::RssRunningSparse<Index_, Value_, Stat_> > runners;
680 sanisizer::reserve(runners, num_blocks);
681 for (std::size_t b = 0; b < num_blocks; ++b) {
682 tatami::resize_container_to_Index_size(nonzeros[b], NR);
683 runners.emplace_back(NR, tmp_mean_ptrs[b], tmp_rss_ptrs[b], nonzeros[b].data());
684 }
685
686 for (Index_ x = 0; x < l; ++x) {
687 auto out = ext->fetch(vbuffer.data(), ibuffer.data());
688 runners[block[x + s]].add(out.number, out.value, out.index);
689 }
690
691 for (std::size_t b = 0; b < num_blocks; ++b) {
692 runners[b].finish();
693 }
694
695 } else {
696 auto ext = tatami::consecutive_extractor<false>(mat, false, s, l);
697 auto buffer = tatami::create_container_of_Index_size<std::vector<Value_> >(NR);
698
699 std::vector<quickstats::RssRunningDense<Value_, Stat_> > runners;
700 sanisizer::reserve(runners, num_blocks);
701 for (std::size_t b = 0; b < num_blocks; ++b) {
702 runners.emplace_back(NR, tmp_mean_ptrs[b], tmp_rss_ptrs[b]);
703 }
704
705 for (Index_ x = 0; x < l; ++x) {
706 auto out = ext->fetch(buffer.data());
707 runners[block[x + s]].add(out);
708 }
709
710 for (std::size_t b = 0; b < num_blocks; ++b) {
711 runners[b].finish();
712 }
713 }
714
715 all_partial_count[thread] = std::move(cur_count);
716 if (do_parallel) {
717 (*all_partial_mean)[thread] = std::move(cur_mean);
718 if (thread > 0) {
719 (*all_partial_rss)[thread - 1] = std::move(cur_rss);
720 }
721 }
722 }, NC, num_threads);
723
724 if (!do_parallel) {
725 const auto& cur_count = *(all_partial_count[0]);
726 for (std::size_t b = 0; b < num_blocks; ++b) {
727 quickstats::rss_to_variance(NR, cur_count[b], output[b].variance);
728 }
729
730 } else {
731 const auto& ap_mean = *all_partial_mean;
732 const auto& ap_rss = *all_partial_rss;
733
734 assert(sanisizer::is_equal(num_blocks, block_sizes.size()));
735 for (int u = 0; u < nused; ++u) {
736 const auto& cur_count = *(all_partial_count[u]);
737 for (std::size_t b = 0; b < num_blocks; ++b) {
738 block_sizes[b] += cur_count[b];
739 }
740 }
741
742 // Computing the global mean.
743 for (std::size_t b = 0; b < num_blocks; ++b) {
744 const auto cur_output = output[b].mean;
745 for (int u = 0; u < nused; ++u) {
746 const auto& cur_mean = (*(ap_mean[u]))[b];
747 const Stat_ mult = static_cast<Stat_>((*(all_partial_count[u]))[b]) / static_cast<Stat_>(block_sizes[b]);
748 for (Index_ d = 0; d < NR; ++d) {
749 cur_output[d] += cur_mean[d] * mult;
750 }
751 }
752 }
753
754 // Combining the RSS. We need to use the safe variant of recenter_rss(), just to protect against the
755 // case where a block has no observations within a particular thread.
756 for (std::size_t b = 0; b < num_blocks; ++b) {
757 const auto& cur_global = output[b].mean;
758 const auto cur_output = output[b].variance;
759 for (int u = 0; u < nused; ++u) {
760 const auto cur_count = (*(all_partial_count[u]))[b];
761 const auto& cur_mean = (*(ap_mean[u]))[b];
762 if (u == 0) {
763 for (Index_ d = 0; d < NR; ++d) {
764 cur_output[d] = quickstats::recenter_rss(cur_count, cur_output[d], cur_mean[d], cur_global[d]);
765 }
766 } else {
767 const auto& cur_rss = (*(ap_rss[u - 1]))[b];
768 for (Index_ d = 0; d < NR; ++d) {
769 cur_output[d] += quickstats::recenter_rss(cur_count, cur_rss[d], cur_mean[d], cur_global[d]);
770 }
771 }
772 }
773 quickstats::rss_to_variance(NR, block_sizes[b], cur_output);
774 }
775 }

845template<typename Value_, typename Index_, typename Block_, typename Stat_>
846void model_gene_variances_blocked(
847 const tatami::Matrix<Value_, Index_>& mat,
848 const Block_* const block,
849 const std::size_t num_blocks,
850 const ModelGeneVariancesBlockedBuffers<Stat_>& buffers,
851 const ModelGeneVariancesOptions& options
852) {
853 auto block_sizes = sanisizer::create<std::vector<Index_> >(num_blocks);
854 if (mat.prefer_rows()) {
855 model_gene_variances_blocked_direct(mat, block, num_blocks, block_sizes, buffers.per_block, options.num_threads);
856 } else {
857 model_gene_variances_blocked_running(mat, block, num_blocks, block_sizes, buffers.per_block, options.num_threads);
858 }
859
860 FitVarianceTrendWorkspace<Stat_> work;
861 auto fopt = options.fit_variance_trend_options;
862 fopt.num_threads = options.num_threads;
863 bool all_trends_fitted = true;
864
865 const auto NR = mat.nrow();
866 for (std::size_t b = 0; b < num_blocks; ++b) {
867 const auto& current = buffers.per_block[b];
868 if (current.fitted == NULL || current.residual == NULL) {
869 all_trends_fitted = false;
870 continue;
871 }
872 if (block_sizes[b] >= 2) {
873 fit_variance_trend(NR, current.mean, current.variance, current.fitted, current.residual, work, fopt);
874 } else {
875 std::fill_n(current.fitted, NR, std::numeric_limits<double>::quiet_NaN());
876 std::fill_n(current.residual, NR, std::numeric_limits<double>::quiet_NaN());
877 }
878 }
879
880 const auto ave_means = buffers.average.mean;
881 const auto ave_variances = buffers.average.variance;
882 const auto ave_fitted = buffers.average.fitted;
883 const auto ave_residuals = buffers.average.residual;
884
885 if ((ave_fitted || ave_residuals) && !all_trends_fitted) {
886 throw std::runtime_error("cannot compute average fitted values/residuals without per-block trend fits");
887 }
888
889 std::vector<Stat_*> tmp_pointers;
890 tmp_pointers.reserve(num_blocks);
891
892 if (options.block_average_policy == BlockAveragePolicy::MEAN) {
893 const auto block_weight = scran_blocks::compute_weights<Stat_>(block_sizes, options.block_weight_policy, options.variable_block_weight_parameters);
894 std::vector<Stat_> tmp_weights;
895 tmp_weights.reserve(num_blocks);
896
897 if (ave_means) {
898 extract_blocked_weights(num_blocks, block_weight, block_sizes, static_cast<Index_>(1), tmp_weights);
899 extract_blocked_pointers(num_blocks, buffers.per_block, block_sizes, static_cast<Index_>(1), [](const auto& x) -> Stat_* { return x.mean; }, tmp_pointers);
900 scran_blocks::parallel_weighted_means(NR, tmp_pointers, tmp_weights.data(), ave_means, /* skip_nan = */ false);
901 }
902
903 // Skip blocks without enough cells to compute the variance.
904 extract_blocked_weights(num_blocks, block_weight, block_sizes, static_cast<Index_>(2), tmp_weights);
905
906 if (ave_variances) {
907 extract_blocked_pointers(num_blocks, buffers.per_block, block_sizes, static_cast<Index_>(2), [](const auto& x) -> Stat_* { return x.variance; }, tmp_pointers);
908 scran_blocks::parallel_weighted_means(NR, tmp_pointers, tmp_weights.data(), ave_variances, /* skip_nan = */ false);
909 }
910
911 if (ave_fitted) {
912 extract_blocked_pointers(num_blocks, buffers.per_block, block_sizes, static_cast<Index_>(2), [](const auto& x) -> Stat_* { return x.fitted; }, tmp_pointers);
913 scran_blocks::parallel_weighted_means(NR, tmp_pointers, tmp_weights.data(), ave_fitted, /* skip_nan = */ false);
914 }
915
916 if (ave_residuals) {
917 extract_blocked_pointers(num_blocks, buffers.per_block, block_sizes, static_cast<Index_>(2), [](const auto& x) -> Stat_* { return x.residual; }, tmp_pointers);
918 scran_blocks::parallel_weighted_means(NR, tmp_pointers, tmp_weights.data(), ave_residuals, /* skip_nan = */ false);
919 }
920
921 } else if (options.block_average_policy == BlockAveragePolicy::QUANTILE) {
922 if (ave_means) {
923 extract_blocked_pointers(num_blocks, buffers.per_block, block_sizes, static_cast<Index_>(1), [](const auto& x) -> Stat_* { return x.mean; }, tmp_pointers);
924 scran_blocks::parallel_quantiles(NR, tmp_pointers, options.block_quantile, ave_means, /* skip_nan = */ false);
925 }
926
927 // Again, skip blocks without enough cells to compute the variance.
928
929 if (ave_variances) {
930 extract_blocked_pointers(num_blocks, buffers.per_block, block_sizes, static_cast<Index_>(2), [](const auto& x) -> Stat_* { return x.variance; }, tmp_pointers);
931 scran_blocks::parallel_quantiles(NR, tmp_pointers, options.block_quantile, ave_variances, /* skip_nan = */ false);
932 }
933
934 if (ave_fitted) {
935 extract_blocked_pointers(num_blocks, buffers.per_block, block_sizes, static_cast<Index_>(2), [](const auto& x) -> Stat_* { return x.fitted; }, tmp_pointers);
936 scran_blocks::parallel_quantiles(NR, tmp_pointers, options.block_quantile, ave_fitted, /* skip_nan = */ false);
937 }
938
939 if (ave_residuals) {
940 extract_blocked_pointers(num_blocks, buffers.per_block, block_sizes, static_cast<Index_>(2), [](const auto& x) -> Stat_* { return x.residual; }, tmp_pointers);
941 scran_blocks::parallel_quantiles(NR, tmp_pointers, options.block_quantile, ave_residuals, /* skip_nan = */ false);
942 }
943 }
